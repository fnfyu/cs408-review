•
阵列乘法器（快速乘法器中的一种） 的运算速度最快——可1个时钟内完成运算
•
由 ALU、移位器、寄存器、控制逻辑 组成的乘法电路运算速度较快——通常需多个时钟
才能完成运算

附件：⽤移位运算、加减运算等效实现乘法（可执⾏）
 
 
#include <stdio.h>
/*⽤移位运算、加法运算等效实现32bit⽆符号数乘法*/
unsigned int multiply_unsigned(unsigned int x, unsigned int y) {
    unsigned int result = 0;
    for (int i = 0; i < 32; i++) {
        // 提取乘数 y 的最低位
        unsigned int bit = y & 1;
        // 如果当前位为 1，加上被乘数 x
        if (bit) {
            result += x;
        }
        // 被乘数 x 左移⼀位，乘数 y 右移⼀位
        x <<= 1;
        y >>= 1;
    }
    return result;
}
int main() {
    unsigned int x, y;
    printf("王道友，请输⼊两个⽆符号整数 x 和 y：");
    scanf("%u %u", &x, &y);
    unsigned int product = multiply_unsigned(x, y);
    printf("结果：%u\n", product);
    return 0;
}
